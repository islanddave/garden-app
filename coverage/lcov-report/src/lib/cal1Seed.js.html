// CAL-1 seed generator — PURE transform: harvest-weights-v2.json -> fail-closed, batch-idempotent seed SQL.
// No DB access at generate time; the fail-CLOSED cultivar keying happens IN-DATABASE at apply time (a
// sample is inserted only if EXACTLY ONE live cultivar matches (crop_type_slug, display_name); 0 or >=2
// matches skip). Pairs with scripts/cal1/gen-cal1-seed.mjs (the CLI). Lockstep with migrations/v4-cal1-pervariety-001.
// (V4-CAL1HARV-001, crucible-hardened V100.)
 
import { UNIT_VOCAB, deriveCultivarWeight } from './cal1Weights.js'
 
export const MODEL_SCHEMA = 'garden.harvest_weights.v2'
 
// Reject a model that isn't the v2 authoring shape (schema-contract versioning: tested reject path).
export function validateModel(model) {
  if (!model || typeof model !== 'object') throw new Error('cal1Seed: model is not an object')
  if (model.schema !== MODEL_SCHEMA) {
    throw new Error(`cal1Seed: unexpected schema "${model.schema}" (want "${MODEL_SCHEMA}")`)
  }
  if (!/^2\./.test(String(model.schema_version || ''))) {
    throw new Error(`cal1Seed: unsupported schema_version "${model.schema_version}" (want 2.x)`)
  }
  if (!Array.isArray(model.by_cultivar_samples)) {
    throw new Error('cal1Seed: by_cultivar_samples must be an array')
  }
  return true
}
 
// A sample is REAL (not the template / not half-authored) iff it has a matchable key + a valid measurement.
function isRealSample(s) {
  if (!s || typeof s !== 'object') return false
  const slug = String(s.crop_type_slug || '')
  const name = String(s.variety_name || '')
  if (!slug || !name || name.startsWith('__TEMPLATE__')) return false
  if (!UNIT_VOCAB.includes(s.unit)) return false
  const tg = Number(s.total_grams)
  const uc = Number(s.unit_count)
  if (!Number.isFinite(tg) || !Number.isFinite(uc) || tg <= 0 || uc <= 0) return false
  if (!s.sampled_at) return false
  return true
}
 
// Collect the real, seedable samples from the model (drops template/null/invalid rows).
export function collectSamples(model) {
  validateModel(model)
  return model.by_cultivar_samples.filter(isRealSample).map((s) => ({
    slug: String(s.crop_type_slug),
    name: String(s.variety_name),
    unit: String(s.unit),
    total_grams: Number(s.total_grams),
    unit_count: Number(s.unit_count),
    sampled_at: String(s.sampled_at),
    note: s.note == null ? null : String(s.note),
  }))
}
 
// Oracle-driven preview: what each (slug,name,unit) group WOULD derive, so the operator sees the numbers
// before applying. Does NOT resolve cultivar_id (that is the DB's fail-closed job at apply time).
export function derivedPreview(model, { minN } = {}) {
  const groups = new Map()
  for (const s of collectSamples(model)) {
    const key = `${s.slug}${s.name}${s.unit}`
    if (!groups.has(key)) groups.set(key, { slug: s.slug, name: s.name, unit: s.unit, samples: [] })
    groups.get(key).samples.push({ total_grams: s.total_grams, unit_count: s.unit_count })
  }
  return [...groups.values()].map((g) => ({
    slug: g.slug,
    name: g.name,
    unit: g.unit,
    derived: deriveCultivarWeight(g.samples, minN ? { minN } : undefined),
  }))
}
 
function sqlStr(s) {
  return `'${String(s).replace(/'/g, "''")}'`
}
function sqlNum(n) {
  const x = Number(n)
  if (!Number.isFinite(x)) throw new Error(`cal1Seed: non-finite numeric ${n}`)
  return String(x)
}
function sqlStrOrNull(s) {
  return s == null ? 'NULL' : sqlStr(s)
}
 
// Emit the fail-closed, batch-idempotent seed SQL + a coverage query + stats. `batch` = the seed_batch for
// this run (a run = a batch). Re-applying the same batch does not double-insert (NOT EXISTS guard).
export function generateSeedSQL(model, { batch, createdBy = 'cal1-seed' } = {}) {
  if (!batch) throw new Error('cal1Seed: a batch id is required (a run = a seed_batch)')
  const samples = collectSamples(model)
  const header =
    '-- 0d-seed-samples.sql — GENERATED by scripts/cal1/gen-cal1-seed.mjs from src/data/harvest-weights-v2.json\n' +
    `-- Batch: ${batch}. Fail-CLOSED keying (insert only if EXACTLY ONE live cultivar matches\n` +
    '-- (crop_type_slug, display_name)) + batch-idempotent (re-applying this batch does not double-insert).\n' +
    '-- Depends on v4-cal1-pervariety-001/0a. Dave-gated apply. Run 0d-coverage.sql to see key resolution.\n'
 
  if (samples.length === 0) {
    return {
      seedSQL: `${header}-- (no real samples in the authoring model yet — nothing to seed)\n`,
      coverageSQL: '-- 0d-coverage.sql — (no real samples yet)\n',
      stats: { samples: 0, distinctKeys: 0, batch },
    }
  }
 
  const valueRows = samples
    .map(
      (s) =>
        `  (${sqlStr(s.slug)}, ${sqlStr(s.name)}, ${sqlStr(s.unit)}, ${sqlNum(s.total_grams)}::numeric, ` +
        `${sqlNum(s.unit_count)}::numeric, ${sqlStr(s.sampled_at)}::timestamptz, ${sqlStrOrNull(s.note)})`,
    )
    .join(',\n')
 
  const seedSQL =
    `${header}\n` +
    'INSERT INTO public.cultivar_weight_sample (cultivar_id, unit, total_grams, unit_count, sampled_at, seed_batch, note, created_by)\n' +
    `SELECT m.id, v.unit, v.total_grams, v.unit_count, v.sampled_at, ${sqlStr(batch)}, v.note, ${sqlStr(createdBy)}\n` +
    `FROM (VALUES\n${valueRows}\n) AS v(slug, name, unit, total_grams, unit_count, sampled_at, note)\n` +
    'JOIN LATERAL (\n' +
    '  SELECT c.id FROM public.cultivar c\n' +
    '   WHERE c.crop_type_slug = v.slug AND c.display_name = v.name AND c.deleted_at IS NULL\n' +
    ') m ON true\n' +
    'WHERE (SELECT count(*) FROM public.cultivar c2\n' +
    '        WHERE c2.crop_type_slug = v.slug AND c2.display_name = v.name AND c2.deleted_at IS NULL) = 1\n' +
    '  AND NOT EXISTS (\n' +
    '    SELECT 1 FROM public.cultivar_weight_sample s\n' +
    `     WHERE s.seed_batch = ${sqlStr(batch)} AND s.cultivar_id = m.id AND s.unit = v.unit\n` +
    '       AND s.sampled_at = v.sampled_at AND s.total_grams = v.total_grams AND s.unit_count = v.unit_count\n' +
    '  );\n'
 
  const keySet = new Map()
  for (const s of samples) keySet.set(`${s.slug}${s.name}`, { slug: s.slug, name: s.name })
  const coverageRows = [...keySet.values()].map((k) => `  (${sqlStr(k.slug)}, ${sqlStr(k.name)})`).join(',\n')
  const coverageSQL =
    '-- 0d-coverage.sql — GENERATED. Run at apply time: which authoring keys resolve to a live cultivar?\n' +
    '-- matches=1 -> seeded; matches=0 -> skipped (no such cultivar); matches>=2 -> skipped (ambiguous name).\n' +
    'SELECT v.slug, v.name,\n' +
    '       (SELECT count(*) FROM public.cultivar c\n' +
    '         WHERE c.crop_type_slug = v.slug AND c.display_name = v.name AND c.deleted_at IS NULL) AS matches\n' +
    `FROM (VALUES\n${coverageRows}\n) AS v(slug, name)\n` +
    'ORDER BY matches, v.slug, v.name;\n'
 
  return { seedSQL, coverageSQL, stats: { samples: samples.length, distinctKeys: keySet.size, batch } }
}
